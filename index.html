const ANTHROPIC_KEY = "sk-ant-YOUR_KEY_HERE"; // ← ضع مفتاحك

function doPost(e) {
  try {
    var body = JSON.parse(e.postData.contents);

    // حفظ في Sheets
    if (body.action === "save") {
      var ss = SpreadsheetApp.openById(body.sheetId);
      var sheet = ss.getActiveSheet();
      sheet.appendRow([
        new Date(), body.tone, body.originalText,
        body.hook, body.rewrite, body.imagePrompt
      ]);
      return respond({status: "saved"});
    }

    // Proxy لـ Anthropic API
    var payload = {
      model: body.model || "claude-sonnet-4-20250514",
      max_tokens: body.max_tokens || 1000,
      system: body.system,
      messages: body.messages
    };

    var res = UrlFetchApp.fetch(
      "https://api.anthropic.com/v1/messages", {
      method: "post",
      contentType: "application/json",
      headers: {
        "x-api-key": ANTHROPIC_KEY,
        "anthropic-version": "2023-06-01"
      },
      payload: JSON.stringify(payload),
      muteHttpExceptions: true
    });

    return respond(JSON.parse(res.getContentText()));

  } catch(err) {
    return respond({error: err.toString()});
  }
}

function respond(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
